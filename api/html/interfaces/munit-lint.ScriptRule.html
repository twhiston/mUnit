    Interface ScriptRule

    A rule of kind script: a JavaScript (CommonJS) module decides what to report, under munit lint and in [mUnit.lint] alike.

    interface ScriptRule {
        id: string;
        level: Level;
        message?: string;
        scope: RuleScope;
        source: string;
        kind: "script";
        module: string;
        options?: unknown;
        check?: ScriptCheck;
        meta?: ScriptMeta;
        project?: ProjectHooks;
        state?: RuleState;
    }
